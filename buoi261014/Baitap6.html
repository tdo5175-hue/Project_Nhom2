Phân Loại Tam Giác (Kiểm Tra Tam Giác Đều)

📌
Yêu cầu đề bài: Nhập vào 3 cạnh của một tam giác. Kiểm tra xem đó có phải là tam giác đều không (3 cạnh bằng nhau).
⚡ Khu vực thao tác trực tiếp
Bộ 3 cạnh test:
TAM GIÁC ĐỀU
• Nhận diện: Cả 3 cạnh bằng nhau: a = b = c = 6.
• Ba góc trong tam giác này đều bằng chính xác 60°.
• Thỏa mãn bất đẳng thức tam giác: 6 + 6 > 6.
>_ KẾT QUẢ THỰC THI (CONSOLE LOGGER)
Ba cạnh [6, 6, 6] => TAM GIÁC ĐỀU
🧠 Tư duy logic & Mã nguồn chi tiết
Baitap6.js 
// =========================================================
// BÀI 6: PHÂN LOẠI TAM GIÁC (KIỂM TRA TAM GIÁC ĐỀU)
// =========================================================

// Bước 1: Nhập độ dài 3 cạnh a, b, c từ người dùng qua prompt
let aStr = prompt("Nhập độ dài cạnh thứ nhất (a):");
let bStr = prompt("Nhập độ dài cạnh thứ hai (b):");
let cStr = prompt("Nhập độ dài cạnh thứ ba (c):");

// Bước 2: Kiểm tra hủy bỏ
if (aStr === null || bStr === null || cStr === null) {
    console.log("Người dùng đã hủy thao tác.");
} else {
    // Bước 3: Ép kiểu dữ liệu chuỗi sang số thực
    let a = parseFloat(aStr);
    let b = parseFloat(bStr);
    let c = parseFloat(cStr);

    // Bước 4: Kiểm tra tính hợp lệ cơ bản
    if (isNaN(a) || isNaN(b) || isNaN(c)) {
        console.error("Lỗi: Cả 3 cạnh phải là số!");
        alert("Vui lòng nhập số hợp lệ cho cả 3 cạnh!");
    } else {
        // Bước 5: Kiểm tra Bất Đẳng Thức Tam Giác (điều kiện hình thành tam giác)
        // 1. Cả 3 cạnh phải dương: a > 0 && b > 0 && c > 0
        // 2. Tổng 2 cạnh bất kỳ luôn lớn hơn cạnh còn lại
        let laTamGiac = (a > 0 && b > 0 && c > 0) &&
                        (a + b > c) &&
                        (a + c > b) &&
                        (b + c > a);

        if (!laTamGiac) {
            console.warn(`Ba cạnh (${a}, ${b}, ${c}) không tạo thành tam giác!`);
            alert(`Ba cạnh (${a}, ${b}, ${c}) KHÔNG tạo thành một tam giác hợp lệ!`);
        } else if (a === b && b === c) {
            // Bước 6: Kiểm tra tam giác đều (3 cạnh bằng nhau: a === b && b === c)
            console.log(`=> TAM GIÁC ĐỀU (a = b = c = ${a})`);
            alert(`=> Đây là TAM GIÁC ĐỀU (vì 3 cạnh bằng nhau: a = b = c = ${a}).`);
        } else {
            // Không phải tam giác đều
            console.log(`=> KHÔNG PHẢI tam giác đều.`);
            alert(`=> Đây KHÔNG PHẢI là tam giác đều.`);
        }
    }
}
