Chuyển Đổi Tiền Tệ Đơn Giản (USD → VND)

📌
Yêu cầu đề bài: Viết mã JS cho phép nhập số lượng USD và tính ra số tiền VND tương ứng (giả sử tỉ giá 25.000). Kết quả hiện ra console.
⚡ Khu vực thao tác trực tiếp
Tỉ giá quy ước trong bài: 1 USD = 25.000 VNĐ
Chọn nhanh:
2.500.000 VNĐ
• Số tiền USD: $100
• Tỉ giá áp dụng: 1 USD = 25.000 VNĐ
• Phép toán thực hiện: 100 × 25.000 = 2.500.000
>_ KẾT QUẢ IN RA CONSOLE (LOG TRỰC TIẾP)
100 USD = 2.500.000 VND
🧠 Tư duy logic & Mã nguồn chi tiết
Baitap1.js 
// =========================================================
// BÀI 1: CHUYỂN ĐỔI TIỀN TỆ ĐƠN GIẢN (USD -> VND)
// =========================================================

// Bước 1: Khai báo tỉ giá cố định (1 USD = 25.000 VND)
// Sử dụng 'const' vì giá trị này không đổi trong suốt bài toán
const TY_GIA = 25000;

// Bước 2: Nhập số lượng USD từ bàn phím bằng prompt
let inputUSD = prompt("Nhập số lượng USD cần đổi sang VND:");

// Bước 3: Kiểm tra nếu người dùng nhấn nút Hủy (Cancel)
if (inputUSD === null) {
    console.log("Người dùng đã hủy thao tác quy đổi.");
} else {
    // Bước 4: Ép kiểu dữ liệu chuỗi từ prompt sang số thực
    let soUSD = parseFloat(inputUSD);

    // Bước 5: Kiểm tra tính hợp lệ của số tiền (không phải chữ và không âm)
    if (isNaN(soUSD) || soUSD < 0) {
        console.error("Lỗi: Số tiền USD không hợp lệ hoặc nhỏ hơn 0!");
        alert("Vui lòng nhập số tiền USD hợp lệ lớn hơn hoặc bằng 0!");
    } else {
        // Bước 6: Tính ra số tiền VND tương ứng
        let soVND = soUSD * TY_GIA;

        // Bước 7: Xuất kết quả ra console theo đúng yêu cầu đề bài
        // Sử dụng toLocaleString("vi-VN") để có dấu chấm phân cách hàng nghìn
        console.log(`$${soUSD} USD = ${soVND.toLocaleString("vi-VN")} VND`);

        // Thông báo thêm qua alert để người dùng dễ quan sát
        alert(`Kết quả: ${soUSD} USD = ${soVND.toLocaleString("vi-VN")} VNĐ`);
    }
}
