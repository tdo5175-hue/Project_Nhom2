Kiểm Tra Số Chẵn / Số Lẻ

📌
Yêu cầu đề bài: Nhập một số bất kỳ từ bàn phím. Thông báo cho người dùng biết đó là số chẵn hay số lẻ.
⚡ Khu vực thao tác trực tiếp
Quy ước toán học: Tính chẵn/lẻ chỉ áp dụng cho tập hợp số nguyên (ℤ)
Thử nhanh:
SỐ CHẴN
• Giá trị kiểm tra: 8
• Biểu thức tính dư: 8 % 2 = 0
• Kết luận: Số 8 chia hết cho 2 nên là SỐ CHẴN.
>_ KẾT QUẢ THỰC THI (CONSOLE LOGGER)
Số 8 là SỐ CHẴN (chia hết cho 2).
🧠 Tư duy logic & Mã nguồn chi tiết
Baitap2.js 
// =========================================================
// BÀI 2: KIỂM TRA SỐ CHẴN / SỐ LẺ
// =========================================================

// Bước 1: Nhận giá trị nhập vào từ bàn phím bằng prompt
let input = prompt("Nhập một số nguyên bất kỳ từ bàn phím:");

// Bước 2: Kiểm tra nếu người dùng bấm Hủy (Cancel)
if (input === null) {
    console.log("Người dùng đã hủy thao tác kiểm tra.");
} else {
    // Bước 3: Chuyển chuỗi nhập vào sang kiểu số nguyên
    let n = Number(input.trim());

    // Bước 4: Kiểm tra tính hợp lệ (phải là số và phải là số nguyên)
    if (input.trim() === "" || isNaN(n) || !Number.isInteger(n)) {
        console.error("Lỗi: Dữ liệu nhập vào phải là một số nguyên!");
        alert("Vui lòng nhập một số nguyên hợp lệ (Ví dụ: -4, 0, 7)!");
    } else {
        // Bước 5: Áp dụng cấu trúc rẽ nhánh if - else với toán tử chia dư (%)
        if (n % 2 === 0) {
            // Chia cho 2 dư 0: Là SỐ CHẴN (áp dụng đúng cho cả số âm)
            let ketQua = `Số ${n} là SỐ CHẴN.`;
            console.log(ketQua);
            alert(ketQua);
        } else {
            // Chia cho 2 không dư 0: Là SỐ LẺ
            let ketQua = `Số ${n} là SỐ LẺ.`;
            console.log(ketQua);
            alert(ketQua);
        }
    }
}
