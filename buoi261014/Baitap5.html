Máy Tính Giảm Giá (Discount Calculator)

📌
Yêu cầu đề bài: Nhập số tiền hóa đơn. Nếu số tiền > 500.000đ thì giảm giá 10%, ngược lại không giảm. Tính số tiền cuối cùng phải trả.
⚡ Khu vực thao tác trực tiếp
Chính sách: Hóa đơn > 500.000 VNĐ được giảm 10% (tiết kiệm ngay)
Hóa đơn mẫu:
675.000 VNĐ
• Tiền hóa đơn gốc: 750.000 VNĐ
• Mức giảm áp dụng: 10% (Tiết kiệm được -75.000 VNĐ)
• Số tiền thực tế phải trả: 750.000 - 75.000 = 675.000 VNĐ
>_ KẾT QUẢ THỰC THI (CONSOLE LOGGER)
Hóa đơn: 750.000đ | Giảm: 10% (-75.000đ) => Thực trả: 675.000đ
🧠 Tư duy logic & Mã nguồn chi tiết
Baitap5.js 
// =========================================================
// BÀI 5: MÁY TÍNH GIẢM GIÁ (DISCOUNT CALCULATOR)
// =========================================================

// Bước 1: Khai báo ngưỡng xét giảm giá và tỉ lệ giảm
// Hóa đơn trên 500.000đ thì giảm 10%
const NGUONG_GIAM = 500000;
const TI_LE_GIAM = 0.10; // 10%

// Bước 2: Nhập số tiền hóa đơn từ người dùng
let inputAmount = prompt("Nhập số tiền hóa đơn (VNĐ):");

// Bước 3: Kiểm tra nếu người dùng nhấn Hủy
if (inputAmount === null) {
    console.log("Người dùng đã hủy tính tiền hóa đơn.");
} else {
    // Bước 4: Ép kiểu dữ liệu chuỗi sang số thực
    let tongTien = parseFloat(inputAmount);

    // Bước 5: Kiểm tra tính hợp lệ (tiền không được là chữ và không âm)
    if (isNaN(tongTien) || tongTien < 0) {
        console.error("Lỗi: Số tiền hóa đơn không hợp lệ!");
        alert("Vui lòng nhập số tiền hóa đơn hợp lệ (>= 0đ)!");
    } else {
        // Bước 6: Áp dụng cấu trúc rẽ nhánh if - else tính tiền giảm
        let tienGiam = 0;
        let phanTram = 0;

        if (tongTien > NGUONG_GIAM) {
            // Hóa đơn > 500.000đ: Được giảm 10%
            phanTram = 10;
            tienGiam = tongTien * TI_LE_GIAM;
        } else {
            // Ngược lại: Không giảm giá (0đ)
            phanTram = 0;
            tienGiam = 0;
        }

        // Bước 7: Tính số tiền thực tế khách hàng phải trả
        let tienPhaiTra = tongTien - tienGiam;

        // Bước 8: Xuất chi tiết hóa đơn ra console và thông báo
        console.log(`Hóa đơn: ${tongTien.toLocaleString("vi-VN")} VND`);
        console.log(`Giảm giá: ${phanTram}% (-${tienGiam.toLocaleString("vi-VN")} VND)`);
        console.log(`Số tiền phải trả: ${tienPhaiTra.toLocaleString("vi-VN")} VND`);

        alert(`HÓA ĐƠN THANH TOÁN:\n• Tiền gốc: ${tongTien.toLocaleString("vi-VN")} VNĐ\n• Giảm giá: ${phanTram}%\n• Tiền phải trả: ${tienPhaiTra.toLocaleString("vi-VN")} VNĐ`);
    }
}
